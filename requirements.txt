# ── Web search (Feature 20) ───────────────────────────────────────────────────
# ddgs (DuckDuckGo, no API key) is imported lazily INSIDE core/websearch.py —
# importing the app or running the whole suite needs neither ddgs nor network.
ddgs                      # Keyless search provider; enables config/websearch.yaml

# ── Long-term memory (Feature 2) ──────────────────────────────────────────────
# Local, on-disk vector memory (facts + daily summaries) with semantic search.
# NOTE: chromadb pulls a fairly large tree (grpcio, kubernetes, opentelemetry,
# uvicorn, ...). It is only imported when enabled in config/memory.yaml, so the
# companion runs unchanged without it.
chromadb                  # Local vector store (embeddings + nearest-neighbour search)

# ── Screen vision (Feature 4) ─────────────────────────────────────────────────
# On-demand screen capture: one frame on hotkey, encoded IN MEMORY (never written
# to disk). mss is pure Python (py3-none-any); Pillow encodes/downscales the PNG.
# Only imported when enabled in config/vision.yaml.
mss                       # Screen capture (active window's monitor work area)
pillow                    # In-memory PNG encoding + downscale
